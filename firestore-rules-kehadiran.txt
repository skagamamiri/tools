// Blok ini sudah digabungkan ke dalam peraturan Firestore penuh (Firebase Console → Firestore → Rules).
// Ia menggunakan fungsi sedia ada isActiveTeacher(), isReviewer() dan isAdmin().
    // ==========================================
    // KEHADIRAN KELAS (IMBAS KOD QR)
    // ==========================================
    match /kehadiranKelas/{recordId} {
      // Guru baca rekod sendiri; Admin + Pentadbir baca semua (laporan).
      allow read: if isActiveTeacher()
        && (
          isReviewer()
          || resource.data.email == request.auth.token.email
        );

      // Guru hanya mencipta rekod sendiri; tidak boleh diubah selepas disimpan.
      allow create: if isActiveTeacher()
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.status in ['HADIR', 'LEWAT', 'LUAR_JADUAL'];

      // Hanya Admin boleh membetulkan atau memadam rekod.
      allow update, delete: if isAdmin();
    }
